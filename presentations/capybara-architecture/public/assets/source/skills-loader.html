  56. def parseSkill(content: String): Result[SkillMd, String] =
  57. content match

  59. yaml.as[SkillMd] match

  75. def loadFrom(dir: Path): Result[Skill, String] = result {
  76. val skillMd = dir / "SKILL.md"

  91. val interfaceCode = readSafely(interfaceFile).ok

  93. Skill(

  97. interface = interfaceCode,

  107. /** Dependency order is also factory-construction order. Only declared safe APIs form compile classpaths. */
  108. def resolve(requested: Seq[String], available: Seq[Skill], builtins: Skill,
  109. includeApis: Boolean = false): Result[Vector[Skill], String] =
  110. Result.catchException { case error => error.getMessage } {
  111. val known = (available :+ builtins).groupBy(_.universe).view.mapValues(_.head).toMap
  112. val completed = scala.collection.mutable.Set.empty[String]
  113. val active = scala.collection.mutable.Set.empty[String]
  114. val ordered = Vector.newBuilder[Skill]
  115. def visit(name: String): Unit =
  116. if !completed(name) then
  117. require(!active(name), s"Skill dependency cycle at: $name")
  118. val skill = known.getOrElse(name, throw IllegalArgumentException(s"Universe not found: $name"))
  119. require(skill.name == skill.universe, s"Skill name must match universe: $name")
  120. active += name
  121. skill.requires.foreach(visit)
  122. if includeApis then skill.apiDependencies.filter(known.contains).foreach(visit)
  123. active -= name
  124. completed += name
  125. if skill.requiresRuntimeClasspath && name != builtins.universe then ordered += skill
  126. requested.foreach(visit)
  127. val skills = ordered.result()
  128. if !includeApis then
  129. val _ = SkillScaffold.bindings(builtins, skills)
  130. skills
  131. }
  133. def loadAll(dir: Path): Seq[Skill] =
  134. // Todo: errors are swallowed
