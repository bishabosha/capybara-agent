  204. private def elapsedSeconds: Long =

  207. private def updateStatus(label: String, force: Boolean = false): Unit =

  227. private def recordChunk(label: String, chars: Int): Unit =
  228. val chunks = receivedChunks.incrementAndGet()

  232. private def checkThinkingDone(): Unit =

  237. override def onChunk(chunk: Chunk[ScalaToolCall]): Unit =
  238. seenAny.set(true)
  239. var chunkId = -1
  240. chunk match {
  241. case Chunk.ResponseItems(_) => ()
  242. case Chunk.Content(content) =>
  243. recordChunk("receiving content", content.length)
  244. checkThinkingDone()

  246. log.print(Console.BOLD + content + Console.RESET)
  247. case Chunk.Thinking(thinking) =>
  248. recordChunk("thinking", thinking.length)
  249. log.print(Console.CYAN + thinking + Console.RESET)
  250. case Chunk.Tools(tool_calls) =>
  251. recordChunk(
  252. "calling tool",
  253. tool_calls.map(call => call.arguments.scala_code.length).sum
  254. )
  255. checkThinkingDone()
  256. if seenTools.compareAndSet(false, true) then log.print("-----\n")
  257. chunkId =
  258. // later we retrieve based on the offset from the first tool call
  259. commandCounter.get() + 1
  260. for scalaCode <- tool_calls do
  261. // todo: extract tool call id from llm?
  262. val promise = Promise[ScalaToolResult]()
  263. val callId = commandCounter.incrementAndGet()

  265. outstandingToolPromises.incrementAndGet()
  266. updateStatus("tool promise started", force = true)
  267. promise.future.onComplete { _ =>
  268. outstandingToolPromises.decrementAndGet()
  269. updateStatus("tool promise completed", force = true)
  270. }
  271. val result =
  272. Skills.resolve(scalaCode.arguments.requestedUniverses, availableSkills, builtinSkill) match
  273. case Result.Ok(skills) =>
  274. ReplExec
  275. .runCodeHarness(
  276. builtinSkill,
  277. skills,
  278. scalaCode.arguments.scala_code,
  279. callId,
  280. log,
  281. cancellationToken
  282. )
  283. case Result.Err(error) =>
  284. Future.successful(ScalaToolResult.Failure(error))
  285. promise.tryCompleteWith(result)
  286. log.print(
  287. s"${Console.YELLOW}${scalaCode.name}[${scalaCode.arguments.requestedUniverses.mkString(",")}]:${Console.RESET}\n"
  288. )
  289. log.print(
  290. s"${Console.YELLOW}```scala${Console.RESET}\n"
  291. )
  292. log.print(s"${Console.YELLOW}${scalaCode.arguments.scala_code}${Console.RESET}")
  293. log.print(s"\n${Console.YELLOW}```${Console.RESET}\n")
  294. }
  295. collected.add((chunkId, chunk))
  297. override def onComplete(
  298. finalState: ChunkOutputState & ChunkFinalState,
  299. usage: OllamaClient.Usage

  301. updateStatus("complete", force = true)
  302. if seenAny.get() then log.print("\n")
  303. chunks.success(
  304. finalState match
  305. case ChunkOutputState.Done =>
  306. (
  307. Result.Ok(RawAgentResponse(collected.asScala.toVector, usage)),

  310. case ChunkOutputState.Error(msg) =>
  311. (Result.Err(msg), scalaCommands.asScala.toMap)
  312. )
  313. }
  314. request.send(chunker).flatMap {
  315. case Result.Ok(_) =>
  316. for

  318. results <- Future.sequence(cs.scalaCommands.map((k, p) => p.future.map(r => (k, r))))
  319. yield cs.collected.map({ response =>
  320. AgentResponse(

  322. chunk match
  323. case Chunk.ResponseItems(items) => ResolvedChunk.ResponseItems(items)
  324. case Chunk.Content(content) => ResolvedChunk.Content(content)
  325. case Chunk.Thinking(thinking) => ResolvedChunk.Thinking(thinking)
  326. case Chunk.Tools(tool_calls) =>
  327. ResolvedChunk.Tools(
  328. tool_calls.zipWithIndex.map { case (tc, idx) =>
  329. (
  330. tc,
  331. results.toMap
  332. .getOrElse(id + idx, ScalaToolResult.Failure("No response"))
  333. )
  334. }.toVector
  335. )
  336. }),
  337. response.usage
  338. )
  339. })
  340. case err: Result.Err[?] => Future.successful(err)
  341. }

  356. private def messagesFromChunks(
  357. chunks: Seq[ScalaResolvedChunk]
  358. ): Vector[OllamaClient.ChatMessage] = {
  359. val thinking = new StringBuilder
  360. val content = new StringBuilder
  361. val toolCalls = Vector.newBuilder[OllamaClient.ToolCall[ScalaToolCall]]
  362. val toolResults = Vector.newBuilder[OllamaClient.ChatMessage]
  363. var responseItems: Option[Vector[ujson.Value]] = None
  365. chunks.foreach {
  366. case ResolvedChunk.ResponseItems(items) =>
  367. responseItems = Some(items)
  368. case ResolvedChunk.Thinking(chunk) =>
  369. thinking.append(chunk)
  370. case ResolvedChunk.Content(chunk) =>
  371. content.append(chunk)
  372. case ResolvedChunk.Tools(calls) =>
  373. calls.foreach { case (toolCall, result) =>
  374. toolCalls += toolCall
  375. toolResults += OllamaClient.ChatMessage.tool(toolCall.name, result.encodeAsJson, toolCall.id)
  376. }
  377. }
  379. val accumulatedToolCalls = toolCalls.result()
  380. val assistant =
  381. if thinking.nonEmpty || content.nonEmpty || accumulatedToolCalls.nonEmpty || responseItems.nonEmpty then
  382. Vector(
  383. OllamaClient.ChatMessage.assistant(
  384. thinking = thinking.toString,
  385. content = content.toString,
  386. toolCalls = accumulatedToolCalls
  387. ).copy(responseItems = responseItems)
  388. )
  389. else Vector.empty
  391. assistant ++ toolResults.result()
  392. }
