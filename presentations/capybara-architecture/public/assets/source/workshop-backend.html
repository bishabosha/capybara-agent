  120. def parent_fd(self, rel):
  121. # Every component is opened relative to an already pinned directory descriptor.
  122. # Reject symlinks, devices/FIFOs and hard links rather than following path races.
  123. fd = os.open(self.root, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
  124. try:
  125. info = os.fstat(fd)
  126. if (info.st_dev, info.st_ino) != (self.state["device"], self.state["inode"]):
  127. raise Denied("WORKSPACE_CHANGED")
  128. for part in rel.split("/")[:-1]:
  129. child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
  130. os.close(fd)
  131. fd = child
  132. return fd
  133. except BaseException:
  134. os.close(fd)

  137. def read(self, rel):
  138. if not isinstance(rel, str) or rel not in READS:
  139. raise Denied("PATH_DENIED")
  140. parent = None
  141. fd = None
  142. try:
  143. parent = self.parent_fd(rel)
  144. fd = os.open(rel.split("/")[-1], os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=parent)
  145. info = os.fstat(fd)

  147. raise Denied("PATH_DENIED")
  148. data = os.read(fd, LIMIT + 1).decode("utf-8")
  149. if rel in self.state["frozen"] and data != self.state["frozen"][rel]:
  150. raise Denied("WORKSPACE_CHANGED")
  151. return safe_text(data)
  152. except (OSError, UnicodeError):
  153. raise Denied("PATH_DENIED") from None
  154. finally:
  155. if fd is not None:
  156. os.close(fd)
  157. if parent is not None:
  158. os.close(parent)

  219. class MacSandbox:
  220. def __init__(self, node, snapshot):
  221. executable = shutil.which(node)
  222. if sys.platform != "darwin" or not Path("/usr/bin/sandbox-exec").is_file() or not executable:
  223. raise Denied("SANDBOX_UNAVAILABLE")
  224. self.node = Path(executable).resolve()
  225. self.snapshot = snapshot.resolve()
  226. # Node must be a real executable, not a shell shim relying on extra host access.
  227. with self.node.open("rb") as executable_file:
  228. if executable_file.read(4) not in (b"\xcf\xfa\xed\xfe", b"\xca\xfe\xba\xbe", b"\xca\xfe\xba\xbf"):
  229. raise Denied("SANDBOX_UNAVAILABLE")
  230. quote = lambda x: json.dumps(str(x), ensure_ascii=False)
  231. self.profile = "\n".join([
  232. "(version 1)", "(deny default)",
  233. f"(allow process-exec (literal {quote(self.node)}))",
  234. "(allow sysctl-read)", "(allow mach-lookup)",
  235. "(allow file-read-metadata)",
  236. # dyld/V8 opens the filesystem root itself during startup. This permits
  237. # enumerating /, not reading its descendants or any user's file contents.
  238. '(allow file-read-data (literal "/"))',
  239. '(allow file-read* (subpath "/System") (subpath "/usr/lib") (subpath "/usr/share"))',
  240. f"(allow file-read* (literal {quote(self.node)}) (subpath {quote(self.snapshot)}))",
  241. '(allow file-read* file-write* (literal "/dev/null") (literal "/dev/random") (literal "/dev/urandom"))',
  242. ])
