  9. class GitHubSkillSuite extends munit.FunSuite:
  10. override val munitTimeout = scala.concurrent.duration.Duration(180, "s")
  11. private val token = "SYNTHETIC_HOST_TOKEN_abcdefghijklmnopqrstuvwxyz"
  12. private val initial = "1" * 40
  13. private val tree = "2" * 40
  14. private val committed = "3" * 40
  15. private def loaded(path: String): Skill = Skills.loadFrom(os.pwd / os.RelPath(path)) match
  16. case Result.Ok(value) => value
  17. case Result.Err(error) => fail(error)
  18. private lazy val builtins = loaded("builtin-skills/capybara-builtins")
  20. private class Remote extends AutoCloseable:
  21. val refs = scala.collection.mutable.Map("main" -> initial)
  22. val requests = scala.collection.mutable.ArrayBuffer.empty[(String, String, ujson.Value)]
  23. val pulls = scala.collection.mutable.ArrayBuffer.empty[ujson.Value]
  24. var secretIssue = false
  25. var successChecks = false
  26. val server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0)
  27. server.createContext("/repos/example/lab", exchange => {
  28. try
  29. assertEquals(exchange.getRequestHeaders.getFirst("Authorization"), "Bearer " + token)
  30. val method = exchange.getRequestMethod
  31. val path = exchange.getRequestURI.getPath.stripPrefix("/repos/example/lab")
  32. val raw = new String(exchange.getRequestBody.readAllBytes(), UTF_8)
  33. val body = if raw.isEmpty then ujson.Null else ujson.read(raw)
  34. requests.synchronized { val _ = requests += ((method, path, body)) }
  35. val issue = ujson.Obj("number" -> 1, "title" -> "Fix whitespace λ", "body" ->
  36. (if secretIssue then token else "Handle empty input, repeated spaces and \\t."))
  37. val (status, result) = (method, path) match
  38. case ("GET", "/issues") => (200, ujson.Arr(issue))
  39. case ("GET", "/issues/1") => (200, issue)
  40. case ("GET", p) if p.startsWith("/git/ref/heads/") =>
  41. refs.get(p.stripPrefix("/git/ref/heads/")) match
  42. case Some(sha) => (200, ujson.Obj("object" -> ujson.Obj("sha" -> sha)))
  43. case None => (404, ujson.Obj("message" -> token))
  44. case ("POST", "/git/refs") =>
  45. val name = body("ref").str.stripPrefix("refs/heads/")
  46. if refs.contains(name) then (422, ujson.Obj("message" -> token))
  47. else
  48. refs(name) = body("sha").str
  49. (201, ujson.Obj("object" -> ujson.Obj("sha" -> body("sha"))))
  50. case ("GET", p) if p.startsWith("/git/commits/") =>
  51. (200, ujson.Obj("tree" -> ujson.Obj("sha" -> tree)))
  52. case ("POST", "/git/trees") => (201, ujson.Obj("sha" -> tree))
  53. case ("POST", "/git/commits") => (201, ujson.Obj("sha" -> committed))
  54. case ("PATCH", p) if p.startsWith("/git/refs/heads/") =>
  55. assert(!body("force").bool)
  56. refs(p.stripPrefix("/git/refs/heads/")) = body("sha").str
  57. (200, ujson.Obj("object" -> ujson.Obj("sha" -> body("sha"))))
  58. case ("GET", "/pulls") => (200, ujson.Arr.from(pulls))
  59. case ("POST", "/pulls") =>
  60. assert(body("draft").bool)
  61. val item = ujson.Obj("number" -> 2, "draft" -> true,
  62. "head" -> ujson.Obj("ref" -> body("head"), "sha" -> committed,
  63. "repo" -> ujson.Obj("full_name" -> "example/lab")),
  64. "base" -> ujson.Obj("ref" -> "main"))
  65. pulls += item
  66. (201, item)
  67. case ("GET", p) if p.endsWith("/status") =>
  68. (200, ujson.Obj("state" -> "pending", "total_count" -> 0))
  69. case ("GET", p) if p.endsWith("/check-runs") =>
  70. val runs = if successChecks then Seq(ujson.Obj("head_sha" -> committed,
  71. "status" -> "completed", "conclusion" -> "success")) else Nil
  72. (200, ujson.Obj("total_count" -> runs.size, "check_runs" -> ujson.Arr.from(runs)))
  73. case _ => (500, ujson.Obj("message" -> token))
  74. val data = ujson.write(result).getBytes(UTF_8)
  75. exchange.getResponseHeaders.set("Content-Type", "application/json")
  76. exchange.sendResponseHeaders(status, data.length.toLong)
  77. exchange.getResponseBody.write(data)
  78. finally exchange.close()
  79. })
  80. server.start()
  81. val endpoint = s"http://127.0.0.1:${server.getAddress.getPort}"
  82. def close(): Unit = server.stop(0)
  84. private def withSkills(test: (Seq[Skill], os.Path, Remote) => Unit): Unit =
  85. val root = os.temp.dir(prefix = "capybara-github-composed-")
  86. val remote = new Remote
  87. try
  88. os.makeDir.all(root / "src")

  90. val coding = loaded("skills/coding")
  91. val local = coding.copy(bindings = coding.bindings.map(b =>
  92. (name = b.name, tpe = b.tpe, value = s"editing.internal.configured(${ujson.write(root.toString)})")))
  93. val github = loaded("skills/github")
  94. val factory = s"githubskill.internal.configured(\"example/lab\", \"main\", ${ujson.write(token)}, ${ujson.write(remote.endpoint)})"
  95. val online = github.copy(bindings = github.bindings.map(b => (name = b.name, tpe = b.tpe, value = factory)))
  96. assertEquals(online.requires, Seq("coding"))
  97. test(Seq(online, local), root, remote)
  98. finally
  99. remote.close()
  100. os.remove.all(root)
  102. private def run(skills: Seq[Skill], code: String): ReplExec.CodeOutput =
  103. ReplExec.runCode(builtins, skills, code, CancellationToken.Never) match
  104. case Result.Ok(value) =>
  105. assert(!value.stdout.contains(token))
  106. assert(!value.stderr.contains(token))
  107. value
  108. case Result.Err(error) => fail(error)
