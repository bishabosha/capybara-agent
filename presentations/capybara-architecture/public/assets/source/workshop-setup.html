  1. package capybara.agent
  3. import steps.result.Result
  5. /** Host setup only: copy tracked starters and prepare skill builds without invoking factories. */
  6. object WorkshopSetup:
  7. case class Workspace(name: String, root: os.Path, environment: os.Path, created: Boolean)
  9. private[agent] def shellQuote(text: String): String = "'" + text.replace("'", "'\"'\"'") + "'"
  11. def create(repository: os.Path, directory: os.Path): Seq[Workspace] =
  12. val starters = Seq(
  13. "coding" -> (repository / "examples/coding-workspace"),
  14. "github" -> (repository / "presentations/capybara-architecture/workshop/project")
  15. )
  16. // Validate all destinations before copying anything. Existing workspaces and
  17. // edited environment files belong to the participant and are never replaced.
  18. val planned = starters.map { (name, source) =>
  19. require(os.isDir(source), s"Missing workshop starter: $source. Run from the Capybara checkout.")
  20. val target = directory / name
  21. val environment = directory / s"$name.env"
  22. val contents = s"export CAPYBARA_CODING_ROOT=${shellQuote(target.toString)}\n"
  23. require(!os.exists(target) || os.isDir(target), s"Workspace path is not a directory: $target")
  24. require(!os.exists(environment) || (os.isFile(environment) && os.read(environment) == contents),
  25. s"Environment file already contains different settings: $environment. Choose another --directory.")
  26. (name, source, target, environment, contents)
  27. }
  28. os.makeDir.all(directory)
  29. planned.map { (name, source, target, environment, contents) =>
  30. val created = !os.exists(target)
  31. if created then os.copy(source, target)
  32. if !os.exists(environment) then os.write(environment, contents)
  33. Workspace(name, target, environment, created)
  34. }
  36. @mainargs.main(doc = "Prepare both workshop workspaces and skill builds; preserve existing participant edits.")
  37. def main(@mainargs.arg(doc = "Workspace parent; choose a new directory for a fresh workshop")
  38. directory: String = ".agent-runtime/demo-workspaces"): Unit =
  39. val builtins = load("builtin-skills/capybara-builtins")
  40. val skills = Seq(load("skills/coding"), load("skills/github"))
  41. ReplExec.prepareSkills(builtins, skills) match
  42. case Result.Err(error) => throw IllegalStateException(error)
  43. case Result.Ok(_) => ()
  44. val workspaces = create(os.pwd, os.Path(directory, os.pwd))
  45. println("Workshop ready. Skill APIs and implementations are built.")
  46. for workspace <- workspaces do
  47. println(s"${workspace.name}: ${workspace.root} (${if workspace.created then "created" else "kept existing edits"})")
  48. println("\nCoding demo:")
  49. println(s". ${shellQuote(workspaces.head.environment.toString)}")
  50. println("./mill run skill-run --skill coding --file examples/snippets/coding.scala")
  51. println("\nGitHub demo:")
  52. println(s". ${shellQuote(workspaces(1).environment.toString)}")
  53. println("# Set CAPYBARA_GITHUB_REPOSITORY and supply CAPYBARA_GITHUB_TOKEN in your shell.")
  54. println("./mill run skill-run --skill github --file examples/snippets/github-lab.scala")
  55. println("\nSetup creates local files only. It does not contact GitHub or store credentials.")
  57. private def load(path: String): Skill = Skills.loadFrom(os.pwd / os.RelPath(path)) match
  58. case Result.Ok(value) => value
  59. case Result.Err(error) => throw IllegalStateException(error)
