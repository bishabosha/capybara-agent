  1. package capybara.agent
  3. /** The manifest is the bridge from trusted factories to explicit safe parameters. */
  4. object SkillScaffold:
  5. def bindings(builtins: Skill, skill: Skill): Seq[SkillBinding] =
  6. bindings(builtins, Seq(skill))
  8. def bindings(builtins: Skill, skills: Seq[Skill]): Seq[SkillBinding] =
  9. val all = builtins.bindings ++ skills.filter(s => s.requiresRuntimeClasspath && s.universe != builtins.universe)
  10. .distinctBy(_.universe).flatMap(_.bindings)
  11. require(all.map(_.name).distinct.size == all.size, "Duplicate capability parameter names in manifests")
  12. for binding <- all do
  13. require(binding.name.matches("[a-zA-Z_][a-zA-Z0-9_]*"), s"Invalid capability name: ${binding.name}")
  14. require(binding.tpe.nonEmpty && binding.value.nonEmpty, s"Incomplete capability binding: ${binding.name}")
  15. require(builtins.bindings.exists(_.name == "output"), "Builtins must provide the output capability")
  16. all
  18. def definition(name: String, bindings: Seq[SkillBinding], code: String): String =
  19. val parameters = bindings.map(b => s"${b.name}: ${b.tpe}").mkString("(using ", ", ", ")")
  20. s"""def $name$parameters: Unit = {
  21. |$code
  22. |()
  23. |}
  24. |""".stripMargin
  26. def invocation(name: String, bindings: Seq[SkillBinding]): String =
  27. def local(binding: SkillBinding): String = s"__capybara_${binding.name}"
  28. val values = bindings.map(b => s"given ${local(b)}: ${b.tpe} = ${b.value}").mkString("\n")
  29. val arguments = bindings.map(local).mkString("(using ", ", ", ")")
  30. s"""{
  31. |$values
  32. |try { $name$arguments; true }
  33. |catch {
  34. | case scala.util.control.NonFatal(error) =>
  35. | __capybara_output.printErr(builtins.baseReflectiveShowable(error))
  36. | false
  37. |}
  38. |}
  39. |""".stripMargin
  41. def entryPoint(name: String, snippet: String, bindings: Seq[SkillBinding]): String =
  42. s"""def $name(__harnessStdout: java.io.PrintStream, __harnessStderr: java.io.PrintStream): Boolean =
  43. |${invocation(snippet, bindings)}
  44. |""".stripMargin
  46. /** Readable version of exactly the same parameter/factory wiring used by the REPL. */
  47. def preview(builtins: Skill, skill: Skill): String =
  48. preview(builtins, Seq(skill))
  50. def preview(builtins: Skill, skills: Seq[Skill]): String =
  51. val all = bindings(builtins, skills)
  52. val imports = (builtins +: skills).distinctBy(_.universe).flatMap(_.predef).mkString("\n")
  53. s"""// Generated by skill-ide. Edit the original skill sources, not this preview.
  54. |object SkillPreview {
  55. |$imports
  56. |// ReplExec compiles this method separately under safe mode.
  57. |${definition("snippet", all, "// Your snippet runs here, with explicitly supplied capabilities.")}
  58. |// This caller is trusted harness code; factories are rejected in safe mode.
  59. |${entryPoint("run", "snippet", all)}
  60. |}
  61. |""".stripMargin
