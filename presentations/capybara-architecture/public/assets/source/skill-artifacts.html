  1. package capybara.agent
  3. import steps.result.Result
  4. import steps.result.Result.eval.ok
  5. import steps.result.Result.apply as result
  7. /** Only safe interface artifacts are published. Factories and credentials stay in implementation JARs. */
  8. object SkillArtifacts:
  9. val Organization = "dev.capybara.skills"
  10. val Version = "0.1.0-SNAPSHOT"
  12. def coordinate(skill: Skill): String = s"$Organization::${skill.universe}-api:$Version"
  14. def dependencies(sources: String*): Seq[String] =
  15. val directive = "(?m)^\\s*//>\\s+using\\s+(?:dep|deps|dependencies)\\s+([^\\n]+)".r
  16. val localApi = (java.util.regex.Pattern.quote(Organization) +
  17. "(?:::([A-Za-z0-9_-]+)-api:|:([A-Za-z0-9_-]+)-api_3:)([^\\s\\\"']+)").r
  18. sources.flatMap(source => directive.findAllMatchIn(source).flatMap { line =>
  19. localApi.findAllMatchIn(line.group(1)).map { artifact =>
  20. require(artifact.group(3) == Version, s"Local skill API version must be $Version")
  21. Option(artifact.group(1)).getOrElse(artifact.group(2))
  22. }
  23. }).distinct
  25. def publish(skill: Skill, source: os.Path): Unit =
  26. val result = os.proc("scala", "--power", "publish", "local", source.toString,
  27. "-S", ReplExec.ScalaVersion, "-language:experimental.safe", "--organization", Organization,
  28. "--name", s"${skill.universe}-api", "--project-version", Version,
  29. "--sources-jar=false", "--doc=false", "--signer", "none")
  30. .call(check = false, timeout = 120000, stdout = os.Pipe, stderr = os.Pipe)
  31. require(result.exitCode == 0, s"Could not publish safe API ${coordinate(skill)}:\n" +
  32. result.out.text() + result.err.text())
  34. @mainargs.main(doc = "Publish checked skill APIs to local Ivy, with dependencies first; never invoke factories.")
  35. def main(@mainargs.arg(doc = "Skill name or directory; comma-separated selections are supported") skill: String): Unit =
  36. publishLocal(skill) match
  37. case Result.Ok(coordinates) =>
  38. println("Published safe APIs to local Ivy (implementations remain separate).")
  39. println("//> using repository ivy2local")
  40. coordinates.foreach(coordinate => println(s"//> using dep \"$coordinate\""))
  41. case Result.Err(error) =>
  42. scala.Console.err.println(error)
  43. sys.exit(1)
  45. def publishLocal(skillDirectories: String): Result[Seq[String], String] =
  46. Result.catchException { case error => error.toString } {
  47. result {
  48. val builtins = Skills.loadFrom(os.pwd / "builtin-skills/capybara-builtins").ok
  49. val selected = skillDirectories.split(",").toSeq.map(_.trim).map { name =>
  50. Skills.loadFrom(if name.contains("/") then os.Path(name, os.pwd) else os.pwd / "skills" / name).ok
  51. }
  52. val available = selected ++ Skills.loadAll(os.pwd / "skills")
  53. val dependencies = Skills.resolve(selected.map(_.universe), available, builtins, includeApis = true).ok
  54. val order = (selected.filter(_.universe == builtins.universe) ++ dependencies).distinctBy(_.universe)
  55. require(order.nonEmpty, "Select a skill with a public API")
  56. // Validate the normal build without constructing capabilities. Explicit publication
  57. // refreshes Ivy even when runtime JARs are cached or the Ivy artifact was removed.
  58. val _ = ReplExec.prepareSkills(builtins, selected).ok
  59. val staging = os.temp.dir(prefix = "capybara-publish-")
  60. try
  61. for current <- order do
  62. val source = staging / current.universe / "Interface.scala"
  63. os.write(source, current.interface, createFolders = true)
  64. publish(current, source)
  65. finally os.remove.all(staging)
  66. order.map(coordinate)
  67. }
  68. }.flatMap(identity)
