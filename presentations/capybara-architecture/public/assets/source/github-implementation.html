  15. private object Wire:
  16. enum Value:
  17. case Obj(fields: Map[String, Value])
  18. case Arr(elements: Vector[Value])
  19. case Str(value: String)
  20. case Num(value: BigDecimal)
  21. case Bool(value: Boolean)
  22. case Null
  23. def apply(key: String): Value = this match
  24. case Obj(fields) => fields.getOrElse(key, invalid())
  25. case _ => invalid()
  26. def get(key: String): Option[Value] = this match
  27. case Obj(fields) => fields.get(key)
  28. case _ => invalid()
  29. def text: String = this match
  30. case Str(value) => value
  31. case _ => invalid()
  32. def integer: Int = this match
  33. case Num(value) if value.isValidInt => value.toInt
  34. case _ => invalid()
  35. def items: Vector[Value] = this match
  36. case Arr(items) => items
  37. case _ => invalid()
  38. import Value.*
  39. def invalid(): Nothing = throw IllegalStateException("INVALID_RESPONSE")
  40. def obj(fields: (String, Value)*): Value = Obj(fields.toMap)
  41. def string(value: String): Value = Str(value)
  42. def number(value: Int): Value = Num(BigDecimal(value))
  43. def render(value: Value): String = value match
  44. case Obj(fields) => fields.toSeq.sortBy(_._1).map((key, value) => render(Str(key)) + ":" + render(value)).mkString("{", ",", "}")
  45. case Arr(items) => items.map(render).mkString("[", ",", "]")
  46. case Str(text) => "\"" + text.flatMap {
  47. case '"' => "\\\""
  48. case '\\' => "\\\\"
  49. case c if c < ' ' => f"\\u${c.toInt}%04x"
  50. case c => c.toString
  51. } + "\""
  52. case Num(value) => value.toString
  53. case Bool(value) => value.toString
  54. case Null => "null"
  55. def parse(source: String): Value =
  56. var at = 0
  57. def space(): Unit = while at < source.length && " \r\n\t".contains(source(at)) do at += 1
  58. def take(c: Char): Unit =
  59. space()
  60. if at >= source.length || source(at) != c then invalid()
  61. at += 1
  62. def text(): String =
  63. take('"')
  64. val result = new StringBuilder
  65. var done = false
  66. while !done && at < source.length do
  67. val c = source(at)
  68. at += 1
  69. if c == '"' then done = true
  70. else if c == '\\' then
  71. if at >= source.length then invalid()
  72. val escaped = source(at)
  73. at += 1
  74. escaped match
  75. case '"' | '\\' | '/' => result.append(escaped)
  76. case 'b' => result.append('\b')
  77. case 'f' => result.append('\f')
  78. case 'n' => result.append('\n')
  79. case 'r' => result.append('\r')
  80. case 't' => result.append('\t')
  81. case 'u' =>
  82. if at + 4 > source.length || !source.substring(at, at + 4).matches("[0-9a-fA-F]{4}") then invalid()
  83. result.append(Integer.parseInt(source.substring(at, at + 4), 16).toChar)
  84. at += 4
  85. case _ => invalid()
  86. else if c < ' ' then invalid()
  87. else result.append(c)
  88. if !done then invalid()
  89. result.toString
  90. def value(depth: Int): Value =
  91. space()
  92. if depth > 64 || at >= source.length then invalid()
  93. source(at) match
  94. case '"' => Str(text())
  95. case '{' =>
  96. take('{')
  97. space()
  98. var fields = Map.empty[String, Value]
  99. if at < source.length && source(at) != '}' then
  100. var more = true
  101. while more do
  102. val key = text()
  103. if fields.contains(key) then invalid()
  104. take(':')
  105. fields += key -> value(depth + 1)
  106. space()
  107. more = at < source.length && source(at) == ','
  108. if more then at += 1
  109. take('}')
  110. Obj(fields)
  111. case '[' =>
  112. take('[')
  113. space()
  114. val items = Vector.newBuilder[Value]
  115. if at < source.length && source(at) != ']' then
  116. var more = true
  117. while more do
  118. items += value(depth + 1)
  119. space()
  120. more = at < source.length && source(at) == ','
  121. if more then at += 1
  122. take(']')
  123. Arr(items.result())
  124. case _ =>
  125. val token = source.substring(at).takeWhile(c => !" \r\n\t,]}".contains(c))
  126. at += token.length
  127. token match
  128. case "true" => Bool(true)
  129. case "false" => Bool(false)
  130. case "null" => Null
  131. case n if n.matches("-?(0|[1-9][0-9]*)(\\.[0-9]+)?([eE][+-]?[0-9]+)?") => Num(BigDecimal(n))
  132. case _ => invalid()
  133. val result = value(0)
  134. space()
  135. if at != source.length then invalid()
  136. result
  138. private class GitHubImpl(repository: String, base: String, token: String, endpoint: String) extends githubskill.GitHub:
  139. import Wire.*
  140. import Wire.Value.*
  141. private class PolicyError(code: String) extends IllegalStateException(code)
  142. private def requirePolicy(ok: Boolean, code: String): Unit = if !ok then throw PolicyError(code)
  143. requirePolicy(repository.matches("[A-Za-z0-9][A-Za-z0-9_.-]*/[A-Za-z0-9][A-Za-z0-9_.-]*"), "INVALID_REPOSITORY")
  144. requirePolicy(base.matches("[A-Za-z0-9][A-Za-z0-9_./-]*") && !base.contains(".."), "INVALID_BRANCH")
  145. requirePolicy(token.length >= 16 && token.length <= 4096 && !token.exists(_.isWhitespace), "AUTH_FAILED")
  146. private val forbidden = Vector(token, Base64.getEncoder.encodeToString(token.getBytes(UTF_8)),
  147. token.getBytes(UTF_8).iterator.map(b => f"${b & 0xff}%02x").mkString)
  148. private val secret = "(?s).*(LAB_ONLY_CANARY_|github_pat_[A-Za-z0-9_]+|gh[pousr]_[A-Za-z0-9_]{16,}|-----BEGIN (?:[A-Z]+ )?PRIVATE KEY-----|sk-[A-Za-z0-9_-]{16,}).*".r
  149. private def publicText(text: String, limit: Int = 1048576): String =
  150. requirePolicy(text.getBytes(UTF_8).length <= limit && !forbidden.exists(text.contains) &&
  151. !secret.matches(text) && !text.exists(c => c < ' ' && !"\n\r\t".contains(c)), "CONTENT_DENIED")
  152. text
  153. private val noProxy = new ProxySelector:
  154. def select(uri: URI): java.util.List[Proxy] = java.util.List.of(Proxy.NO_PROXY)
  155. def connectFailed(uri: URI, address: SocketAddress, error: java.io.IOException): Unit = ()
  156. private val client = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(20))
  157. .followRedirects(HttpClient.Redirect.NEVER).proxy(noProxy).build()
  158. private def request(method: String, suffix: String, body: Value = Null): Value =
  159. val payload = if body == Null then "" else publicText(render(body), 2097152)
  160. val publisher = if body == Null then HttpRequest.BodyPublishers.noBody()
  161. else HttpRequest.BodyPublishers.ofString(payload, UTF_8)
  162. val req = HttpRequest.newBuilder(URI.create(endpoint + "/repos/" + repository + suffix))
  163. .timeout(Duration.ofSeconds(30)).header("Authorization", "Bearer " + token)
  164. .header("Accept", "application/vnd.github+json").header("Content-Type", "application/json")
  165. .header("X-GitHub-Api-Version", "2026-03-10").header("User-Agent", "capybara-agent")
  166. .method(method, publisher).build()
  167. try
  168. val response = client.send(req, HttpResponse.BodyHandlers.ofInputStream())
  169. val input = response.body()
  170. try
  171. requirePolicy(Set(200, 201)(response.statusCode()), response.statusCode() match
  172. case 401 => "AUTH_FAILED"
  173. case 404 => "NOT_FOUND"
  174. case 409 | 422 => "CONFLICT"
  175. case 429 => "RATE_LIMITED"
  176. case _ => "GITHUB_REQUEST_REJECTED")
  177. val bytes = input.readNBytes(8388609)
  178. requirePolicy(bytes.length <= 8388608, "INVALID_RESPONSE")
  179. parse(new String(bytes, UTF_8))
  180. finally input.close()
  181. catch
  182. case error: PolicyError => throw error
  183. case error: IllegalStateException if error.getMessage == "INVALID_RESPONSE" => throw error
  184. case NonFatal(_) => throw PolicyError("GITHUB_UNAVAILABLE")
  185. private def validSha(value: String): String =
  186. requirePolicy(value.matches("[0-9a-f]{40}"), "INVALID_RESPONSE")
  187. value
  188. private def topic(name: String): String =
  189. requirePolicy(name.startsWith("capybara/") && name != base &&
  190. name.matches("[A-Za-z0-9][A-Za-z0-9_/-]*") && !name.contains("//"), "INVALID_BRANCH")
  191. name
  192. private def head(name: String): String = validSha(request("GET", "/git/ref/heads/" + name)("object")("sha").text)
  193. private def current(branch: githubskill.Branch): Unit =
  194. requirePolicy(head(topic(branch.name)) == validSha(branch.commit), "BRANCH_CHANGED")
  196. def lookupIssue(number: Int): githubskill.Issue =
  197. requirePolicy(number > 0, "INVALID_REQUEST")
  198. val item = request("GET", s"/issues/$number")
  199. requirePolicy(item("number").integer == number && item.get("pull_request").isEmpty, "INVALID_RESPONSE")
  200. val body = item.get("body").filter(_ != Null).map(_.text).getOrElse("")
  201. githubskill.Issue(number, publicText(item("title").text, 4096), publicText(body),
  202. s"https://github.com/$repository/issues/$number")
  204. def listIssues(page: Int): Seq[githubskill.IssueSummary] =
  205. requirePolicy(page > 0 && page <= 1000, "INVALID_REQUEST")
  206. val items = request("GET", s"/issues?state=open&per_page=25&page=$page").items
  207. requirePolicy(items.size <= 25, "INVALID_RESPONSE")
  208. items.filter(_.get("pull_request").isEmpty).map { item =>
  209. val n = item("number").integer
  210. requirePolicy(n > 0, "INVALID_RESPONSE")
  211. githubskill.IssueSummary(n, publicText(item("title").text, 4096), s"https://github.com/$repository/issues/$n")
  212. }
  214. def branch(name: String): githubskill.Branch = githubskill.Branch(topic(name), head(topic(name)))
  216. def createBranch(name: String): githubskill.Branch =
  217. topic(name)
  218. val commit = head(base)
  219. val _ = request("POST", "/git/refs", obj("ref" -> Str("refs/heads/" + name), "sha" -> Str(commit)))
  220. githubskill.Branch(name, commit)
  222. def commit(branch: githubskill.Branch, message: String, paths: Seq[String])
  223. (using files: editing.TextFiles): githubskill.Branch =
  224. publicText(message, 4096)
  225. requirePolicy(message.trim.nonEmpty && paths.nonEmpty && paths.size <= 64 && paths.distinct.size == paths.size, "INVALID_REQUEST")
  226. val changes = paths.map { path =>
  227. requirePolicy(!path.startsWith(".") && !path.split('/').exists(p => p.isEmpty || p == ".." || p.startsWith(".")), "PATH_DENIED")
  228. val content = publicText(files.read(path))
  229. obj("path" -> Str(path), "mode" -> Str("100644"), "type" -> Str("blob"), "content" -> Str(content))
  230. }
  231. // Validate every file and the full payload before any remote write.
  232. val patch = Arr(changes.toVector)
  233. publicText(render(patch), 2097152)
  234. current(branch)
  235. val baseTree = validSha(request("GET", "/git/commits/" + branch.commit)("tree")("sha").text)
  236. val tree = validSha(request("POST", "/git/trees", obj("base_tree" -> Str(baseTree), "tree" -> patch))("sha").text)
  237. val commit = validSha(request("POST", "/git/commits", obj("message" -> Str(message),
  238. "tree" -> Str(tree), "parents" -> Arr(Vector(Str(branch.commit)))))("sha").text)
  239. val _ = request("PATCH", "/git/refs/heads/" + branch.name, obj("sha" -> Str(commit), "force" -> Bool(false)))
  240. githubskill.Branch(branch.name, commit)
  242. def createPullRequest(branch: githubskill.Branch, title: String, body: String): githubskill.PullRequest =
  243. publicText(title, 256)
  244. publicText(body)
  245. requirePolicy(title.trim.nonEmpty, "INVALID_REQUEST")
  246. current(branch)
  247. val owner = repository.takeWhile(_ != '/')
  248. val existing = request("GET", s"/pulls?state=open&head=$owner:${branch.name}&base=$base&per_page=100").items
  249. val matching = existing.filter(item => item("head")("ref").text == branch.name &&
  250. item("head")("sha").text == branch.commit && item("base")("ref").text == base &&
  251. item("head")("repo")("full_name").text.equalsIgnoreCase(repository))
  252. requirePolicy(matching.size <= 1, "INVALID_RESPONSE")
  253. val item = matching.headOption.getOrElse(request("POST", "/pulls", obj("title" -> Str(title),
  254. "body" -> Str(body), "head" -> Str(branch.name), "base" -> Str(base),
  255. "draft" -> Bool(true), "maintainer_can_modify" -> Bool(false))))
  256. val number = item("number").integer
  257. requirePolicy(number > 0, "INVALID_RESPONSE")
  258. githubskill.PullRequest(number, s"https://github.com/$repository/pull/$number")
  260. def checks(branch: githubskill.Branch): githubskill.CheckSummary =
  261. current(branch)
  262. val statuses = request("GET", s"/commits/${branch.commit}/status?per_page=100")
  263. val checks = request("GET", s"/commits/${branch.commit}/check-runs?per_page=100")
  264. val runs = checks("check_runs").items
  265. requirePolicy(checks("total_count").integer == runs.size && statuses("total_count").integer <= 100, "INVALID_RESPONSE")
  266. val count = statuses("total_count").integer
  267. val failed = Set("failure", "error")(statuses("state").text) || runs.exists(r =>
  268. r.get("conclusion").filter(_ != Null).exists(c => Set("failure", "cancelled", "timed_out", "action_required")(c.text)))
  269. val passed = (runs.nonEmpty || count > 0) && (count == 0 || statuses("state").text == "success") &&
  270. runs.forall(r => r("head_sha").text == branch.commit && r("status").text == "completed" && r("conclusion").text == "success")
  271. githubskill.CheckSummary(branch.commit, if failed then "failed" else if passed then "passed" else "pending")
