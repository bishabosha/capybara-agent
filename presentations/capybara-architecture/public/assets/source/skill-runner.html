  1. package capybara.agent
  3. import steps.result.Result

  7. /** Direct developer entry point. Does not initialize ScalaAgent or a model provider. */
  8. object SkillRunner:
  9. @mainargs.main(doc = "Execute a Scala snippet directly through ReplExec, without an LLM.")
  10. def main(
  11. @mainargs.arg(doc = "Skill directory, or basic for standard-library code") skill: String,
  12. @mainargs.arg(doc = "Path to a Scala snippet") file: String
  13. ): Unit =
  14. run(skill, file) match
  15. case Result.Ok(output) =>
  16. scala.Console.print(output.stdout)
  17. scala.Console.err.print(output.stderr)
  18. ReplExec.decodeOutput(output) match
  19. case ReplExec.ScalaToolResult.Failure(_, _, _) => sys.exit(1)
  20. case _ => ()
  21. case Result.Err(error) =>
  22. scala.Console.err.println(error)
  23. sys.exit(1)
  25. def run(skillDirectory: String, snippetFile: String): Result[ReplExec.CodeOutput, String] =
  26. Result.catchException { case error => error.toString } {
  27. result {
  28. val builtins = Skills.loadFrom(os.pwd / "builtin-skills" / "capybara-builtins").ok
  29. val selected = skillDirectory.split(",").toSeq.map(_.trim).map { name =>
  30. if name == "basic" then Skills.Basic
  31. else Skills.loadFrom(if name.contains("/") then os.Path(name, os.pwd) else os.pwd / "skills" / name).ok
  32. }
  33. val code = Skills.readSafely(os.Path(snippetFile, os.pwd)).ok
  34. ReplExec.runCode(builtins, selected, code, CancellationToken.Never).ok
  35. }
  36. }.flatMap(identity)
