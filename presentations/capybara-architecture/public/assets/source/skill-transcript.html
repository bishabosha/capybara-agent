  1. package capybara.agent
  3. import steps.result.Result
  5. /** Render generated REPL commands without compiling or executing a model's snippet. */
  6. object SkillTranscript:
  7. private val normalizedId = "00000000_0000_0000_0000_000000000001"
  9. def render(builtins: Skill, skill: Skill, code: String, root: os.Path = os.pwd): String =
  10. def load(selected: Skill): Seq[String] =
  11. if !selected.requiresRuntimeClasspath then Nil
  12. else
  13. val folder = root / ".agent-runtime" / "interface" / selected.universe
  14. Seq(s":jar ${folder / "sigs.jar"}", s":jar ${folder / "impl.jar"}") ++ selected.predef.toSeq
  15. val bindings = SkillScaffold.bindings(builtins, skill)
  16. val method = s"agentCode_$normalizedId"
  17. val entry = s"harnessEntry_$normalizedId"
  18. val commands = load(builtins) ++ (
  19. if skill.requiresRuntimeClasspath && skill.universe != builtins.universe then load(skill) else Nil
  20. ) ++ Seq(
  21. "// HOST: clear the language-feature setting before changing mode.",
  22. ":settings -language:experimental.safe",
  23. SkillScaffold.definition(method, bindings, code),
  24. "// HOST: clear the language-feature setting again.",
  25. ":settings -language:experimental.captureChecking",
  26. SkillScaffold.entryPoint(entry, method, bindings)
  27. )
  28. s"""// Generated command transcript for universe '${skill.universe}'.
  29. |// The per-call UUID is normalized; all methods use the actual scaffold templates.
  30. |// Initial ReplDriver language: experimental.captureChecking.
  31. |// Cache packaging, if needed, happens before these REPL commands.
  32. |
  33. |${commands.mkString("\n\n")}
  34. |// HOST invocation, not a submitted REPL command:
  35. |// invokeEntryPoint("$entry", stdout.stream, stderr.stream)
  36. |// This reflects the compiled entry method with two host-owned PrintStreams.
  37. |// Finally: resetToInitial(), restore initialState, and clear diagnostics.
  38. |""".stripMargin
  40. @mainargs.main(doc = "Print generated REPL commands for one tool call; do not execute it.")
  41. def main(
  42. @mainargs.arg(doc = "Skill directory, or basic") skill: String,
  43. @mainargs.arg(doc = "Snippet file") file: String
  44. ): Unit =
  45. def load(path: String): Skill = Skills.loadFrom(os.Path(path, os.pwd)) match
  46. case Result.Ok(value) => value
  47. case Result.Err(error) => throw IllegalArgumentException(error)
  48. val builtins = load("builtin-skills/capybara-builtins")
  49. val selected = if skill == "basic" then Skills.Basic else load(skill)
  50. scala.Console.print(render(builtins, selected, os.read(os.Path(file, os.pwd))))
