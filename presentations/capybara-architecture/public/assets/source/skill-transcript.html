  1. package capybara.agent
  3. import steps.result.Result
  5. /** Render generated REPL commands without compiling or executing a model's snippet. */
  6. object SkillTranscript:
  7. private val normalizedId = "00000000_0000_0000_0000_000000000001"
  9. def render(builtins: Skill, skill: Skill, code: String, root: os.Path = os.pwd): String =
  10. renderSkills(builtins, Seq(skill), code, root)
  12. def renderSkills(builtins: Skill, selected: Seq[Skill], code: String, root: os.Path = os.pwd): String =
  13. val skills = Skills.resolve(selected.map(_.universe),
  14. selected ++ Skills.loadAll(root / "skills") :+ Skills.Basic, builtins) match
  15. case Result.Ok(value) => value
  16. case Result.Err(error) => throw IllegalArgumentException(error)
  17. def load(selected: Skill): Seq[String] =
  18. if !selected.requiresRuntimeClasspath then Nil
  19. else
  20. val folder = root / ".agent-runtime" / "interface" / selected.universe
  21. Seq(s":jar ${folder / "sigs.jar"}", s":jar ${folder / "impl.jar"}") ++ selected.predef.toSeq
  22. val bindings = SkillScaffold.bindings(builtins, skills)
  23. val method = s"agentCode_$normalizedId"
  24. val entry = s"harnessEntry_$normalizedId"
  25. val apis = Skills.resolve(selected.map(_.universe),
  26. selected ++ Skills.loadAll(root / "skills") :+ Skills.Basic, builtins, includeApis = true) match
  27. case Result.Ok(value) => value
  28. case Result.Err(error) => throw IllegalArgumentException(error)
  29. val apiCommands = apis.map(s => s":jar ${root / ".agent-runtime/interface" / s.universe / "sigs.jar"}")
  30. val implementationCommands = skills.flatMap(s =>
  31. Seq(s":jar ${root / ".agent-runtime/interface" / s.universe / "impl.jar"}") ++ s.predef.toSeq)
  32. val commands = load(builtins) ++ apiCommands ++ implementationCommands ++ Seq(
  33. "// HOST: clear the language-feature setting before changing mode.",
  34. ":settings -language:experimental.safe",
  35. SkillScaffold.definition(method, bindings, code),
  36. "// HOST: clear the language-feature setting again.",
  37. ":settings -language:experimental.captureChecking",
  38. SkillScaffold.entryPoint(entry, method, bindings)
  39. )
  40. s"""// Generated command transcript for skills '${selected.map(_.universe).mkString(",")}'.
  41. |// The per-call UUID is normalized; all methods use the actual scaffold templates.
  42. |// Initial ReplDriver language: experimental.captureChecking.
  43. |// Cache packaging, if needed, happens before these REPL commands.
  44. |
  45. |${commands.mkString("\n\n")}
  46. |// HOST invocation, not a submitted REPL command:
  47. |// invokeEntryPoint("$entry", stdout.stream, stderr.stream)
  48. |// This reflects the compiled entry method with two host-owned PrintStreams.
  49. |// Finally: resetToInitial(), restore initialState, and clear diagnostics.
  50. |""".stripMargin
  52. @mainargs.main(doc = "Print generated REPL commands for one tool call; do not execute it.")
  53. def main(
  54. @mainargs.arg(doc = "Skill directory, or basic") skill: String,
  55. @mainargs.arg(doc = "Snippet file") file: String
  56. ): Unit =
  57. def load(path: String): Skill = Skills.loadFrom(os.Path(path, os.pwd)) match
  58. case Result.Ok(value) => value
  59. case Result.Err(error) => throw IllegalArgumentException(error)
  60. val builtins = load("builtin-skills/capybara-builtins")
  61. val selected = skill.split(",").toSeq.map(_.trim).map { name =>
  62. if name == "basic" then Skills.Basic
  63. else load(if name.contains("/") then name else s"skills/$name")
  64. }
  65. scala.Console.print(renderSkills(builtins, selected, os.read(os.Path(file, os.pwd))))
