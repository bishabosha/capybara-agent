  1. package capybara.agent
  3. import steps.result.Result
  4. import steps.result.Result.eval.ok
  5. import steps.result.Result.apply as result
  7. /** Set up BSP directly from a skill's sources and Scala CLI directives. No build files. */
  8. object SkillIde:
  9. @mainargs.main(doc = "Prepare an existing skill directory for Metals or IntelliJ using Scala CLI.")
  10. def main(@mainargs.arg(doc = "Skill directory") skill: String): Unit =
  11. prepare(skill) match
  12. case Result.Ok(directory) =>
  13. println(s"Open $directory in Metals or import it as a BSP project in IntelliJ.")
  14. println(s"Capability wiring: ${directory / ".agent-runtime" / "SkillPreview.scala"}")
  15. case Result.Err(error) =>
  16. scala.Console.err.println(error)
  17. sys.exit(1)
  19. def prepare(skillDirectory: String): Result[os.Path, String] =
  20. Result.catchException { case error => error.toString } {
  21. result {
  22. val directory = os.Path(skillDirectory, os.pwd)
  23. val skill = Skills.loadFrom(directory).ok
  24. val builtinsDirectory = os.pwd / "builtin-skills" / "capybara-builtins"
  25. val builtins = Skills.loadFrom(builtinsDirectory).ok
  26. val preview = directory / ".agent-runtime" / "SkillPreview.scala"
  27. // The builtin skill is already present when developing that skill itself.
  28. val baseline =
  29. if directory == builtinsDirectory then Nil
  30. else Seq(
  31. (builtinsDirectory / "scripts" / "Interface.scala").toString,
  32. (builtinsDirectory / "scripts" / "Implementation.scala").toString
  33. )
  34. val selected = if directory == builtinsDirectory then Skills.Basic else skill
  35. val resolved = ReplExec.prepareSkills(builtins, Seq(selected)).ok
  36. val dependencyJars = resolved.filterNot(_.universe == selected.universe).flatMap(s =>
  37. Seq("--extra-jar", (os.pwd / ".agent-runtime/interface" / s.universe / "impl.jar").toString))
  38. os.write.over(preview, SkillScaffold.preview(builtins, resolved), createFolders = true)
  39. val _ = os.call(
  40. cmd = Seq("scala", "setup-ide", directory.toString, preview.toString) ++ baseline ++
  41. dependencyJars ++ Seq("--workspace", directory.toString, "--scala", ReplExec.ScalaVersion, "--scalac-option=-preview"),
  42. cwd = directory
  43. )
  44. directory
  45. }
  46. }.flatMap(identity)
