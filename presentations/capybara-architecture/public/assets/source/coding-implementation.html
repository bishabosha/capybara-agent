  11. /** JDK-only adapter: skills load in an isolated classloader, without host dependencies. */
  12. private class CodingImpl(root: String, node: String, python: String, hostPath: String)
  13. extends workshop.Coding {
  14. private class PolicyError(code: String) extends IllegalStateException(code)
  15. private val errorCodes = Set(
  16. "WORKSHOP_NOT_CONFIGURED", "WORKSPACE_CHANGED", "PATH_DENIED", "WRITE_DENIED",
  17. "CONTENT_DENIED", "CONTENT_CHANGED", "SANDBOX_UNAVAILABLE", "INVALID_REQUEST",
  18. "WORKSHOP_BACKEND_FAILED"
  19. )
  21. private def invoke(operation: String, arguments: String*): String =
  22. var child: Process | Null = null
  23. try
  24. if arguments.exists(_.getBytes(UTF_8).length > 16384) then throw PolicyError("CONTENT_DENIED")
  25. val helper = Path.of("").toAbsolutePath.resolve("scripts/workshop.py").toString
  26. val builder = new ProcessBuilder(python, "-I", "-S", helper,
  27. "rpc-lines", "--repo", root, "--node", node)
  28. builder.environment().clear()
  29. val _ = builder.environment().put("PATH", hostPath)
  30. val _ = builder.redirectError(ProcessBuilder.Redirect.DISCARD)
  31. val process = builder.start()
  32. child = process
  33. val reader = CompletableFuture.supplyAsync(() => process.getInputStream.readNBytes(65537))
  34. val fields = operation +: arguments.toVector
  35. val input = fields.map(s => Base64.getEncoder.encodeToString(s.getBytes(UTF_8))).mkString("", "\n", "\n")
  36. val out = process.getOutputStream
  37. try out.write(input.getBytes(UTF_8))
  38. finally out.close()
  39. if !process.waitFor(25, TimeUnit.SECONDS) then throw PolicyError("WORKSHOP_BACKEND_FAILED")
  40. val bytes = reader.get(1, TimeUnit.SECONDS)
  41. if process.exitValue() != 0 || bytes.length > 65536 then throw PolicyError("WORKSHOP_BACKEND_FAILED")
  42. val response = new String(bytes, UTF_8).split("\n", -1)
  43. if response.length != 3 || response(2).nonEmpty then throw PolicyError("WORKSHOP_BACKEND_FAILED")
  44. if response(0) == "ERROR" then
  45. throw PolicyError(if errorCodes(response(1)) then response(1) else "WORKSHOP_BACKEND_FAILED")
  46. if response(0) != "OK" then throw PolicyError("WORKSHOP_BACKEND_FAILED")
  47. new String(Base64.getDecoder.decode(response(1)), UTF_8)
  48. catch
  49. case error: PolicyError => throw error
  50. case NonFatal(_) => throw PolicyError("WORKSHOP_BACKEND_FAILED")
  51. finally
  52. if child != null then
  53. val descendants = child.descendants()
  54. try descendants.forEach(handle => { val _ = handle.destroyForcibly() })
  55. finally descendants.close()
  56. val _ = child.destroyForcibly()
  58. def read(path: String): String = invoke("read", path)
  60. def replace(path: String, expected: String, next: String): Unit =
  61. val _ = invoke("replace", path, expected, next)
  63. def transformSource(f: String -> String): Unit =
  64. val before = read("src/word-count.mjs")
  65. replace("src/word-count.mjs", before, f(before))
  67. def test(): workshop.TestSummary =
  68. val status = invoke("test")
  69. if !Set("passed", "failed", "timeout")(status) then throw PolicyError("WORKSHOP_BACKEND_FAILED")
  70. workshop.TestSummary(status)
  72. def diff(): String = invoke("diff")
  74. def sandboxChecks(): Seq[String] = invoke("checks").split(",").toVector
  75. }
