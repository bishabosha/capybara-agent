  2. package editing.internal
  4. import scala.language.experimental.captureChecking
  5. import java.nio.charset.StandardCharsets.UTF_8
  6. import java.nio.file.{Files, LinkOption, Path, StandardCopyOption}
  7. import scala.jdk.CollectionConverters.*
  8. import scala.util.control.NonFatal
  10. /** The trusted implementation supplies file effects; snippets receive only TextFiles. */
  11. private class TextFilesImpl(workspace: String) extends editing.TextFiles {
  12. private val root = Path.of(workspace).toRealPath()
  13. private val maximumBytes = 1024 * 1024
  14. private class FileError(code: String) extends IllegalStateException(code)
  16. private def checked[A](action: => A): A =
  17. try action
  18. catch
  19. case error: FileError => throw error
  20. case NonFatal(_) => throw FileError("FILE_OPERATION_FAILED")
  22. private def resolve(name: String): Path =
  23. val relative = Path.of(name)
  24. if relative.isAbsolute || relative.iterator().asScala.exists(_.toString == "..") then
  25. throw FileError("PATH_DENIED")
  26. val path = root.resolve(relative).normalize()
  27. if !path.startsWith(root) then throw FileError("PATH_DENIED")
  28. var component = root
  29. for part <- root.relativize(path).iterator().asScala do
  30. component = component.resolve(part)
  31. if Files.isSymbolicLink(component) then throw FileError("PATH_DENIED")
  32. path
  34. private def regular(path: Path): Unit =
  35. if !Files.isRegularFile(path, LinkOption.NOFOLLOW_LINKS) then throw FileError("NOT_A_TEXT_FILE")
  37. def list(path: String): Seq[String] = checked {
  38. val directory = resolve(path)
  39. val stream = Files.newDirectoryStream(directory)
  40. try stream.iterator().asScala.toVector
  41. .filterNot(Files.isSymbolicLink(_))
  42. .map(root.relativize(_).toString.replace(java.io.File.separatorChar, '/')).sorted
  43. finally stream.close()
  44. }
  46. def read(path: String): String = checked {
  47. val target = resolve(path)
  48. regular(target)
  49. val input = Files.newInputStream(target, LinkOption.NOFOLLOW_LINKS)
  50. val bytes = try input.readNBytes(maximumBytes + 1) finally input.close()
  51. if bytes.length > maximumBytes then throw FileError("FILE_TOO_LARGE")
  52. UTF_8.newDecoder().decode(java.nio.ByteBuffer.wrap(bytes)).toString
  53. }
  55. private def store(path: String, text: String): Unit =
  56. val bytes = text.getBytes(UTF_8)
  57. if bytes.length > maximumBytes then throw FileError("FILE_TOO_LARGE")
  58. val target = resolve(path)
  59. if target == root then throw FileError("PATH_DENIED")
  60. if Files.exists(target, LinkOption.NOFOLLOW_LINKS) then regular(target)
  61. val parent = target.getParent
  62. if !Files.isDirectory(parent, LinkOption.NOFOLLOW_LINKS) then throw FileError("PARENT_NOT_FOUND")
  63. val temporary = Files.createTempFile(parent, ".capybara-edit-", ".tmp")
  64. try
  65. val _ = Files.write(temporary, bytes)
  66. if Files.exists(target) && Files.getFileStore(target).supportsFileAttributeView("posix") then
  67. val _ = Files.setPosixFilePermissions(temporary,
  68. Files.getPosixFilePermissions(target, LinkOption.NOFOLLOW_LINKS))
  69. // A new inode avoids modifying another file through a hard-linked destination.
  70. val _ = Files.move(temporary, target, StandardCopyOption.ATOMIC_MOVE, StandardCopyOption.REPLACE_EXISTING)
  71. finally
  72. val _ = Files.deleteIfExists(temporary)
  74. def write(path: String, text: String): Unit = checked { store(path, text) }
  76. def replace(path: String, expected: String, next: String): Unit = checked {
  77. if read(path) != expected then throw FileError("CONTENT_CHANGED")
  78. store(path, next)
  79. }
  81. def edit(path: String)(transform: String -> String): Unit =
  82. val before = read(path)
  83. replace(path, before, transform(before))
  84. }
  86. @caps.rejectSafe
  87. def impl(): editing.TextFiles =
  88. configured(sys.env.getOrElse("CAPYBARA_CODING_ROOT", Path.of("").toAbsolutePath.toString))
  90. /** Only trusted harness wiring can select the workspace. */
  91. @caps.rejectSafe
  92. def configured(root: String): editing.TextFiles = new TextFilesImpl(root)
