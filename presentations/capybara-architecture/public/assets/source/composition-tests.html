  39. test("a dependent skill uses a locally published safe API and the harness injects both capabilities") {
  40. val root = os.temp.dir(prefix = "capybara-composition-")
  41. try
  42. os.write(root / "note.txt", "hello")
  43. val coding = loaded("skills/coding")
  44. val configured = coding.copy(bindings = coding.bindings.map(b =>
  45. (name = b.name, tpe = b.tpe, value = s"editing.internal.configured(${ujson.write(root.toString)})")))
  46. val adapter = Skill(name = "composition-test", description = "Read through the safe editing API",
  47. universe = "composition-test", interface = """//> using repository ivy2local
  48. |//> using dep "dev.capybara.skills::coding-api:0.1.0-SNAPSHOT"
  49. |package composition
  50. |import scala.language.experimental.captureChecking
  51. |trait ReadNote extends caps.SharedCapability { def note()(using editing.TextFiles): String }
  52. |""".stripMargin,
  53. code = """//> using repository ivy2local
  54. |//> using dep "dev.capybara.skills::coding-api:0.1.0-SNAPSHOT"
  55. |package composition.internal
  56. |import scala.language.experimental.captureChecking
  57. |@caps.rejectSafe
  58. |def impl(): composition.ReadNote = new composition.ReadNote {
  59. | def note()(using files: editing.TextFiles): String = files.read("note.txt")
  60. |}
  61. |""".stripMargin,
  62. api = "Read a note", predef = None,
  63. bindings = Seq((name = "notes", tpe = "composition.ReadNote", value = "composition.internal.impl()")),
  64. requires = Seq("coding"))
  65. val output = ReplExec.runCode(builtins, Seq(adapter, configured),
  66. "coding.edit(\"note.txt\")(_ + \"!\"); puts(notes.note())", CancellationToken.Never) match
  67. case Result.Ok(value) => value
  68. case Result.Err(error) => fail(error)
  69. assert(!output.failed, output.stderr)
  70. assertEquals(ujson.read(output.stdout)("special.success").str, "hello!")
  71. val rejected = ReplExec.runCode(builtins, Seq(adapter, configured),
  72. "coding.edit(\"note.txt\")(text => notes.note())", CancellationToken.Never)
  73. assert(rejected.isInstanceOf[Result.Err[?]])
  74. assertEquals(SkillArtifacts.dependencies(adapter.code), Seq("coding"))
  75. finally os.remove.all(root)
  76. }
