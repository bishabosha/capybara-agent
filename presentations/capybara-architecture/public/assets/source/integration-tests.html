  1. package capybara.agent
  3. import steps.result.Result
  5. /** Exercises the public skill through the same safe compiler and trusted harness as the agent. */
  6. class WorkshopSuite extends munit.FunSuite:
  7. override val munitTimeout = scala.concurrent.duration.Duration(180, "s")
  9. private def loaded(path: String): Skill = Skills.loadFrom(os.pwd / os.RelPath(path)) match
  10. case Result.Ok(skill) => skill
  11. case Result.Err(error) => fail(error)
  13. private lazy val builtins = loaded("builtin-skills/capybara-builtins")
  14. private val python = "/usr/bin/python3"
  15. private lazy val node = os.proc("/usr/bin/which", "node").call().out.text().trim
  17. private def withWorkshop(test: (Skill, os.Path) => Unit): Unit =
  18. val parent = os.temp.dir(prefix = "capybara-workshop-suite-")
  19. val root = parent / "repo"
  20. try
  21. val setup = os.proc(python, "scripts/workshop.py", "init", "--repo", root.toString).call()
  22. assert(ujson.read(setup.out.text())("ok").bool)
  23. val original = loaded("skills/coding")
  24. val factory = s"workshop.internal.configured(${ujson.write(root.toString)}, ${ujson.write(node)}, ${ujson.write(python)})"
  25. val skill = original.copy(bindings = original.bindings.map(b => (name = b.name, tpe = b.tpe, value = factory)))
  26. test(skill, root)
  27. finally os.remove.all(parent)
  29. private def run(skill: Skill, code: String): ReplExec.CodeOutput =
  30. ReplExec.runCode(builtins, skill, code) match
  31. case Result.Ok(output) => output
  32. case Result.Err(error) => fail(error)
  34. test("workshop policy and real sandbox probes") {
  35. assume(System.getProperty("os.name").contains("Mac"))
  36. val result = os.proc(python, "scripts/test_workshop.py").call(check = false, timeout = 60000)
  37. assertEquals(result.exitCode, 0, result.out.text() + result.err.text())

  40. test("coding universe compiles in safe mode and completes the word-count workflow") {
  41. assume(System.getProperty("os.name").contains("Mac"))
  42. withWorkshop { (skill, root) =>
  43. val baseline = run(skill, "puts(coding.test())")
  44. assert(!baseline.failed)
  45. assertEquals(ujson.read(baseline.stdout)("special.success")("status").str, "failed")
  46. val fix = """export function countWords(text) {
  47. | const clean = text.trim();
  48. | return clean === '' ? 0 : clean.split(/\s+/u).length;
  49. |}
  50. |""".stripMargin
  51. val code = s"""val before = coding.read("src/word-count.mjs")
  52. |coding.replace("src/word-count.mjs", before, ${ujson.write(fix)})
  53. |puts(coding.test())
  54. |puts(coding.diff())
  55. |puts(coding.sandboxChecks())
  56. |""".stripMargin
  57. val output = run(skill, code)
  58. assert(!output.failed, output.stderr)
  59. val lines = output.stdout.linesIterator.map(ujson.read(_)("special.success")).toVector
  60. assertEquals(lines.head("status").str, "passed")
  61. assert(lines(1).str.contains("+ const clean"))
  62. assertEquals(lines(2).arr.size, 5)
  63. assertEquals(os.read(root / "src/word-count.mjs"), fix)
  64. }
  65. }
