  1. package capybara.agent
  3. import steps.result.Result
  5. /** Generic text-file capability and capture checking through the production harness. */
  6. class WorkshopSuite extends munit.FunSuite:
  7. override val munitTimeout = scala.concurrent.duration.Duration(180, "s")
  9. private def loaded(path: String): Skill = Skills.loadFrom(os.pwd / os.RelPath(path)) match
  10. case Result.Ok(skill) => skill
  11. case Result.Err(error) => fail(error)
  13. private lazy val builtins = loaded("builtin-skills/capybara-builtins")
  15. private def withWorkspace(test: (Skill, os.Path) => Unit): Unit =
  16. val root = os.temp.dir(prefix = "capybara-text-files-")
  17. try
  18. os.write(root / "Greeting.scala", "val greeting = \"hello, world\"\n")
  19. os.write(root / "README.md", "Original documentation.\n")
  20. val original = loaded("skills/coding")
  21. val factory = s"editing.internal.configured(${ujson.write(root.toString)})"
  22. val skill = original.copy(bindings = original.bindings.map(b =>
  23. (name = b.name, tpe = b.tpe, value = factory)))
  24. test(skill, root)
  25. finally os.remove.all(root)
  27. private def run(skill: Skill, code: String): ReplExec.CodeOutput =
  28. ReplExec.runCode(builtins, skill, code) match
  29. case Result.Ok(output) => output
  30. case Result.Err(error) => fail(error)
  32. test("generic coding API lists, reads, creates and edits files of different formats") {
  33. withWorkspace { (skill, root) =>
  34. val output = run(skill, os.read(os.pwd / "examples/snippets/coding.scala"))
  35. assert(!output.failed, output.stderr)
  36. assertEquals(os.read(root / "Greeting.scala"), "val greeting = \"hello, capybara\"\n")
  37. assertEquals(os.read(root / "notes.txt"), "Edited through the coding capability.\n")
  38. val markdown = run(skill, "coding.edit(\"README.md\")(_.replace(\"Original\", \"Updated\"))")
  39. assert(!markdown.failed, markdown.stderr)
  40. assertEquals(os.read(root / "README.md"), "Updated documentation.\n")
  41. val json = run(skill, "coding.write(\"config.json\", \"{}\\n\"); puts(coding.read(\"config.json\"))")
  42. assert(!json.failed, json.stderr)
  43. assertEquals(ujson.read(json.stdout)("special.success").str, "{}\n")
  44. os.write(root / "run.sh", "#!/bin/sh\n")
  45. os.perms.set(root / "run.sh", "rwxr-xr-x")
  46. val script = run(skill, "coding.edit(\"run.sh\")(_ + \"echo done\\n\")")
  47. assert(!script.failed, script.stderr)
  48. assert(java.nio.file.Files.isExecutable(java.nio.file.Path.of((root / "run.sh").toString)))
  49. }
  50. }
  52. test("workspace boundaries and stale replacement checks are implemented by the file capability") {
  53. withWorkspace { (skill, root) =>
  54. os.symlink(root / "outside", root / os.up)
  55. for (code, expected) <- Seq(
  56. "coding.read(\"../credentials\")" -> "PATH_DENIED",
  57. "coding.read(\"/etc/passwd\")" -> "PATH_DENIED",
  58. "coding.write(\"outside/escape.txt\", \"x\")" -> "PATH_DENIED",
  59. "coding.replace(\"README.md\", \"stale\", \"new\")" -> "CONTENT_CHANGED"
  60. ) do
  61. val output = run(skill, code)
  62. assert(output.failed)
  63. assert(output.stderr.contains(expected))
  64. assertEquals(os.read(root / "README.md"), "Original documentation.\n")
  65. }
  66. }
  68. test("capture checking rejects extra capabilities and safe mode rejects ambient effects") {
  69. withWorkspace { (skill, root) =>
  70. for code <- Seq(
  71. "coding.edit(\"README.md\")(text => { puts(text); text })",
  72. "coding.edit(\"README.md\")(text => coding.read(\"Greeting.scala\"))",
  73. "coding.edit(\"README.md\")(text => { coding.write(\"side.txt\", text); text })",
  74. "editing.internal.impl()",
  75. "puts(sys.env(\"CAPYBARA_API_KEY\"))",
  76. "new java.lang.ProcessBuilder(\"env\").start()",
  77. "puts(java.nio.file.Files.readString(java.nio.file.Path.of(\"/etc/passwd\")))"
  78. ) do
  79. assert(ReplExec.runCode(builtins, skill, code).isInstanceOf[Result.Err[?]], code)
  80. assert(!os.exists(root / "side.txt"))
  81. assertEquals(os.read(root / "README.md"), "Original documentation.\n")
  82. }
  83. }
